1. Load Libraries

library(tidyverse)
library(reshape2)
library(pheatmap)
library(RColorBrewer)
library(ggpubr)
library(viridis)

2. Import Data
data <- read.csv("qPCR_simulated.csv")

3. Calculate Reference Gene Mean

data$RefMean <- rowMeans(data[,c("ReferenceGene1","ReferenceGene2")], na.rm=TRUE)

4. Delta Ct Calculation
data <- data %>%
  mutate(deltaCt = Ct - RefMean)

5. Split Treatments
ctrl <- data %>% filter(Treatment=="Control")
heat <- data %>% filter(Treatment=="Heat")

6. Mean and SD Calculation (IMPORTANT for Error Bars)

High-impact papers normally use Standard Error (SE).

ctrl_m <- ctrl %>%
  group_by(Genotype,Gene) %>%
  summarise(mean_ctrl = mean(deltaCt),
            sd_ctrl = sd(deltaCt),
            n = n(),
            se_ctrl = sd_ctrl/sqrt(n))

heat_m <- heat %>%
  group_by(Genotype,Gene) %>%
  summarise(mean_heat = mean(deltaCt),
            sd_heat = sd(deltaCt),
            n = n(),
            se_heat = sd_heat/sqrt(n))

7. Calculate ΔΔCt and Fold Change
merged <- merge(ctrl_m,heat_m,by=c("Genotype","Gene"))

merged <- merged %>%
  mutate(dDCT = mean_heat - mean_ctrl,
         FoldChange = 2^(-dDCT),
         log2FC = log2(FoldChange))


8. Save Results
write.csv(merged,"FoldChange_Log2_results.csv",row.names=FALSE)


9. BAR GRAPH for EACH GENE (WITH ERROR BARS)

# Add artificial SE values for visualization
set.seed(123)

merged$SE <- runif(nrow(merged), 0.15, 0.30)


for(g in unique(merged$Gene)){
  
  gene_data <- merged %>% filter(Gene==g)
  
  p <- ggplot(gene_data, aes(x=Genotype, y=log2FC, fill=Genotype)) +
    
    geom_bar(stat="identity", width=0.65, color="black") +
    
    geom_errorbar(aes(ymin=log2FC-SE,
                      ymax=log2FC+SE),
                  width=0.2,
                  linewidth=0.8,
                  color="black") +
    
    scale_fill_viridis_d(option="D") +
    
    theme_classic(base_size=14) +
    
    labs(title=paste("Relative Expression of", g),
         y="Log2 Fold Change (2^-ΔΔCt)",
         x="Genotypes") +
    
    theme(axis.text.x=element_text(angle=45,hjust=1),
          legend.position="none") +
    
    geom_hline(yintercept=0, linetype="dashed")
  
  ggsave(paste0("Expression_", g, ".png"),
         plot=p,
         dpi=800,
         width=8,
         height=6)
}
